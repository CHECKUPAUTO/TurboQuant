pub fn pack_3bit(values: &[u8], out: &mut [u8])

Packs 8 × 3-bit values into 3 bytes.

§Examples

use turboquant_core::bitpack::pack_3bit;

let values: [u8; 8] = [0, 1, 2, 3, 4, 5, 6, 7];
let mut out = [0u8; 3];
pack_3bit(&values, &mut out);
