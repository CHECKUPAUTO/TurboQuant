1//! GGUF file parser.
2
3use crate::types::GgufHeader;
4use turboquant_core::error::TurboQuantError;
5
6/// Parse a GGUF file header from a byte slice.
7///
8/// # Errors
9///
10/// Returns an error if the magic number or format version is invalid.
11pub fn parse_header(data: &[u8]) -> Result<GgufHeader, TurboQuantError> {
12    if data.len() < 16 {
13        return Err(TurboQuantError::InvalidGguf(
14            "File too short for GGUF header".into(),
15        ));
16    }
17    // GGUF magic: "GGUF" = 0x47 0x47 0x55 0x46
18    if &data[0..4] != b"GGUF" {
19        return Err(TurboQuantError::InvalidGguf(
20            "Invalid GGUF magic number".into(),
21        ));
22    }
23    let version = u32::from_le_bytes([data[4], data[5], data[6], data[7]]);
24    let tensor_count = u64::from_le_bytes([
25        data[8], data[9], data[10], data[11], data[12], data[13], data[14], data[15],
26    ]);
27    Ok(GgufHeader {
28        version,
29        tensor_count,
30    })
31}
