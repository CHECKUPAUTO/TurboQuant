1//! QJL (Quantized Johnson-Lindenstrauss) 3-bit quantizer.
2//!
3//! Quantizes to 3 bits with a 1-bit residual correction,
4//! guaranteeing that dot products are preserved.
5
6use half::f16;
7use serde::{Deserialize, Serialize};
8
9/// Configuration for the QJL quantizer.
10#[derive(Clone, Debug, Serialize, Deserialize)]
11pub struct QjlConfig {
12    /// Number of bits per value (default 3).
13    pub bits: u8,
14    /// Block size for per-block scaling.
15    pub block_size: usize,
16    /// Scale computation mode.
17    pub scale_mode: ScaleMode,
18    /// Correction mode.
19    pub correction: CorrectionMode,
20}
21
22impl Default for QjlConfig {
23    fn default() -> Self {
24        Self {
25            bits: 3,
26            block_size: 64,
27            scale_mode: ScaleMode::PerBlockAbsMax,
28            correction: CorrectionMode::OneBitResidual {
29                learned_scale: 0.01,
30            },
31        }
32    }
33}
34
35/// How to compute the per-block scale.
36#[derive(Clone, Debug, Serialize, Deserialize)]
37pub enum ScaleMode {
38    /// Use the absolute maximum value per block.
39    PerBlockAbsMax,
40    /// Use the specified percentile per block.
41    #[allow(clippy::cast_precision_loss)]
42    PerBlockPercentile(f32),
43    /// Adaptive: derive from residual variance.
44    Adaptive,
45    /// Fixed scale value.
46    Fixed(f32),
47}
48
49/// Correction mode for residual quantization error.
50#[derive(Clone, Debug, Serialize, Deserialize)]
51pub enum CorrectionMode {
52    /// No correction applied.
53    None,
54    /// 1-bit residual sign correction with learned scale.
55    OneBitResidual {
56        /// Scale factor for the 1-bit correction.
57        learned_scale: f32,
58    },
59}
60
61/// A single compressed block: packed quantized values + scale.
62#[derive(Clone, Debug)]
63pub struct CompressedBlock {
64    /// Packed 3-bit quantized values.
65    pub packed: Vec<u8>,
66    /// Scale for this block (f16).
67    pub scale: f16,
68    /// 1-bit correction signs (packed bits).
69    pub correction_bits: Option<Vec<u8>>,
70}

72/// The QJL quantizer.
73pub struct QjlQuantizer {
74    config: QjlConfig,
75}

77impl QjlQuantizer {
78    /// Create a new QJL quantizer with the given config.
79    #[must_use]
80    pub const fn new(config: QjlConfig) -> Self {
81        Self { config }
82    }
83
84    /// Return the configuration.
85    #[must_use]
86    pub const fn config(&self) -> &QjlConfig {
87        &self.config
88    }
89
90    /// Compute the optimal scale for a block of values.
91    #[must_use]
92    pub fn compute_scale(&self, block: &[f32]) -> f32 {
93        match self.config.scale_mode {
94            ScaleMode::PerBlockAbsMax => {
95                block.iter().map(|x| x.abs()).fold(0.0f32, f32::max) + 1e-8
96            }
97            ScaleMode::PerBlockPercentile(p) => {
98                let mut abs_vals: Vec<f32> = block.iter().map(|x| x.abs()).collect();
99                abs_vals.sort_by(|a, b| a.partial_cmp(b).unwrap_or(std::cmp::Ordering::Equal));
100                let idx = ((abs_vals.len() - 1) as f32 * p).round() as usize;
101                abs_vals[idx.clamp(0, abs_vals.len() - 1)] + 1e-8
102            }
103            ScaleMode::Adaptive => {
104                let mean = block.iter().sum::<f32>() / block.len() as f32;
105                let variance =
106                    block.iter().map(|x| (x - mean).powi(2)).sum::<f32>() / block.len() as f32;
107                variance.sqrt().max(1e-8)
108            }
109            ScaleMode::Fixed(s) => s,
110        }
111    }
112
113    /// Quantize a block of f32 values to 3-bit with optional correction.
114    ///
115    /// # Examples
116    ///
117    /// ```
118    /// use turboquant_core::qjl::{QjlQuantizer, QjlConfig, ScaleMode, CorrectionMode};
119    ///
120    /// let config = QjlConfig {
121    ///     bits: 3,
122    ///     block_size: 64,
123    ///     scale_mode: ScaleMode::PerBlockAbsMax,
124    ///     correction: CorrectionMode::OneBitResidual { learned_scale: 0.01 },
125    /// };
126    /// let quantizer = QjlQuantizer::new(config);
127    /// let data: Vec<f32> = (0..64).map(|i| (i as f32 - 32.0) / 10.0).collect();
128    /// let compressed = quantizer.quantize_block(&data);
129    /// let decompressed = quantizer.dequantize_block(&compressed, 64);
130    /// assert_eq!(decompressed.len(), 64);
131    /// ```
132    #[allow(clippy::cast_possible_truncation, clippy::cast_precision_loss)]
133    #[must_use]
134    pub fn quantize_block(&self, block: &[f32]) -> CompressedBlock {
135        let scale = self.compute_scale(block);
136        let half_range = 3.5_f32; // (8-1)/2 for 3-bit
137        let levels = 8_u8;
138        let increment = 0.5_f32; // 1.0 / (levels / 2)
139
140        // Normalize and quantize to 3-bit levels
141        let mut quantized: Vec<u8> = Vec::with_capacity(block.len());
142        let mut correction_bits: Option<Vec<u8>> = None;
143
144        match self.config.correction {
145            CorrectionMode::None => {
146                for &x in block {
147                    let x_norm = (x / scale).clamp(-1.0, 1.0);
148                    let q_val = (x_norm * half_range / increment).round() * increment;
149                    let q_clamped = q_val.clamp(-half_range, half_range);
150                    // Map to 0..8 range
151                    let idx = ((q_clamped / increment) + half_range / increment).round() as u8;
152                    quantized.push(idx.clamp(0, levels - 1));
153                }
154            }
155            CorrectionMode::OneBitResidual { learned_scale: _ } => {
156                let mut signs = vec![0u8; block.len().div_ceil(8)];
157                for (i, &x) in block.iter().enumerate() {
158                    let x_norm = (x / scale).clamp(-1.0, 1.0);
159                    let q_val = (x_norm * half_range / increment).round() * increment;
160                    let q_clamped = q_val.clamp(-half_range, half_range);
161                    let idx = ((q_clamped / increment) + half_range / increment).round() as u8;
162                    quantized.push(idx.clamp(0, levels - 1));
163
164                    // 1-bit residual correction sign
165                    let residual = x_norm - q_clamped;
166                    if residual > 0.0 {
167                        signs[i / 8] |= 1 << (i % 8);
168                    }
169                }
170                correction_bits = Some(signs);
171            }
172        }
173
174        // Pack 3-bit values
175        use crate::bitpack::pack_3bit_slice;
176        let num_values = block.len();
177        let padded_len = num_values.next_multiple_of(8);
178        let mut padded_quant = quantized;
179        padded_quant.resize(padded_len, 0);
180        let packed = pack_3bit_slice(&padded_quant).expect("quantized values should be packable");
181
182        CompressedBlock {
183            packed,
184            scale: f16::from_f32(scale),
185            correction_bits,
186        }
187    }
188
189    /// Dequantize a compressed block back to f32 values.
190    #[allow(clippy::cast_precision_loss)]
191    #[must_use]
192    pub fn dequantize_block(&self, block: &CompressedBlock, num_values: usize) -> Vec<f32> {
193        use crate::bitpack::unpack_3bit_slice;
194        let scale = block.scale.to_f32();
195        let half_range = 3.5_f32;
196        let increment = 0.5_f32;
197
198        let padded_len = num_values.next_multiple_of(8);
199        let quantized =
200            unpack_3bit_slice(&block.packed, padded_len).expect("packed data should be unpackable");
201
202        quantized
203            .iter()
204            .take(num_values)
205            .enumerate()
206            .map(|(i, &q)| {
207                let q_float = (f32::from(q) - half_range / increment) * increment;
208                let mut val = q_float / half_range * scale;
209
210                // Apply 1-bit correction if available
211                if let Some(ref signs) = block.correction_bits {
212                    if let CorrectionMode::OneBitResidual { learned_scale } = self.config.correction
213                    {
214                        let sign_bit = (signs[i / 8] >> (i % 8)) & 1;
215                        let correction = if sign_bit == 1 {
216                            learned_scale
217                        } else {
218                            -learned_scale
219                        };
220                        val += correction * scale;
221                    }
222                }
223
224                val
225            })
226            .collect()
227    }

241    #[test]
242    fn test_scale_per_block_abs_max() {
243        let config = QjlConfig {
244            scale_mode: ScaleMode::PerBlockAbsMax,
245            ..Default::default()
246        };
247        let quantizer = QjlQuantizer::new(config);
248        let scale = quantizer.compute_scale(&[-5.0, 3.0, 1.0, 0.0]);
249        assert!((scale - 5.0).abs() < 1e-6);
250    }

265        let config = QjlConfig {
266            scale_mode: ScaleMode::Adaptive,
267            ..Default::default()

269        let quantizer = QjlQuantizer::new(config);

276    #[test]
277    fn test_quantize_dequantize_no_correction() {
278        let config = QjlConfig {
279            correction: CorrectionMode::None,
280            ..Default::default()
281        };
282        let quantizer = QjlQuantizer::new(config);
283        let data: Vec<f32> = (0..64).map(|i| (i as f32 - 32.0) / 10.0).collect();
284        let compressed = quantizer.quantize_block(&data);
285        let decompressed = quantizer.dequantize_block(&compressed, 64);
286
287        assert_eq!(decompressed.len(), 64);
288        // Values should be non-zero
289        let sum: f32 = decompressed.iter().map(|x| x.abs()).sum();
290        assert!(sum > 0.0);
291    }

293    #[test]
294    fn test_quantize_dequantize_with_correction() {
295        let config = QjlConfig {
296            correction: CorrectionMode::OneBitResidual {
297                learned_scale: 0.01,
298            },
299            ..Default::default()
300        };
301        let quantizer = QjlQuantizer::new(config);
302        let data: Vec<f32> = (0..128).map(|i| (i as f32 - 64.0) / 20.0).collect();
303        let compressed = quantizer.quantize_block(&data);
304        let decompressed = quantizer.dequantize_block(&compressed, 128);
305
306        // With correction, reconstruction should be reasonable
307        let mse: f32 = data
308            .iter()
309            .zip(decompressed.iter())
310            .map(|(a, b)| (a - b).powi(2))
311            .sum::<f32>()
312            / data.len() as f32;
313
314        let signal_power: f32 = data.iter().map(|x| x.powi(2)).sum::<f32>() / data.len() as f32;
315        let snr = 10.0 * (signal_power / mse.max(1e-10)).log10();
316        assert!(snr > 2.0, "SNR too low: {snr} dB");
317    }
