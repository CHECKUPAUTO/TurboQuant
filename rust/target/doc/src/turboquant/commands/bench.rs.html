3use colored::Colorize;
4use turboquant_cpu::CpuBackend;
5
6/// Run the bench command.
7///
8/// # Errors
9///
10/// Returns an error if benchmarking fails.
11pub fn run(
12    head_dim: usize,
13    seq_len: usize,
14    num_heads: usize,
15    _iterations: usize,
16) -> Result<(), Box<dyn std::error::Error>> {
17    println!("{}", "TurboQuant Benchmark".bold().cyan());
18    println!("{}", "═══════════════════".bold());
19    println!();
20    println!(
21        "Configuration: head_dim={}, seq_len={}, num_heads={}",
22        head_dim.to_string().cyan(),
23        seq_len.to_string().cyan(),
24        num_heads.to_string().cyan()
25    );
26    println!();
27
28    let backend = CpuBackend::new();
29    let results = backend.full_benchmark(head_dim, seq_len, num_heads);
30
31    println!("{}", "Results:".bold().green());
32    println!(
33        "  FP16 memory (K+V):     {:>8.2} MB",
34        results.fp16_total_mb.to_string().yellow()
35    );
36    println!(
37        "  TurboQuant K cache:    {:>8.2} MB",
38        results.compressed_k_mb.to_string().yellow()
39    );
40    println!(
41        "  TurboQuant V cache:    {:>8.2} MB",
42        results.compressed_v_mb.to_string().yellow()
43    );
44    println!(
45        "  Compression ratio:     {:>8.2}x",
46        results.compression_ratio.to_string().bright_green().bold()
47    );
48    println!(
49        "  Throughput:            {:>8.2} GB/s",
50        results.compression_throughput_gbps.to_string().yellow()
51    );
52    println!(
53        "  Elapsed:               {:>8.2} ms",
54        results.elapsed_ms.to_string().yellow()
55    );
56
57    if results.compression_ratio >= 5.0 {
58        println!(
59            "\n{} Target compression ratio achieved (≥ 5.0×)",
60            "✓".green().bold()
61        );
62    } else {
63        println!(
64            "\n{} Compression ratio below target ({} < 5.0×)",
65            "⚠".yellow().bold(),
66            format!("{:.1}", results.compression_ratio).yellow()
67        );
68    }
69
70    Ok(())
71}
